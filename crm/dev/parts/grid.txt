/* ---------- Speaker und Lektoren: Excel-Ansicht ---------- */
const XL={sort:{k:'n',dir:1},hidden:(()=>{try{return JSON.parse(localStorage.getItem('cetl-crm-hidden')||'[]')}catch(_){return []}})(),menu:false};
const saveHidden=()=>{try{localStorage.setItem('cetl-crm-hidden',JSON.stringify(XL.hidden))}catch(_){}};
const gridCols=()=>SF.filter(f=>f.w&&!XL.hidden.includes(f.k));
function cellHtml(s,f){
  const v=s[f.k];
  if(v==null||v==='')return'';
  if(f.k==='con')return `<span class="pill ${SCON[v]?SCON[v][1]:'partner'}">${esc(SCON[v]?SCON[v][0]:v)}</span>`;
  if(f.k==='v')return `<span class="pill ${SSTAT[v]||'partner'}">${esc(v)}</span>`;
  if(f.k==='prio')return `<b class="prio-${esc(String(v).charAt(0))}">${esc(v)}</b>`;
  if(f.t==='num')return Number(v).toLocaleString('de-AT');
  if(f.k==='em')return `<a href="mailto:${esc(v)}" tabindex="-1">${esc(v)}</a>`;
  return esc(v);
}
function sortVal(s,k){const f=SF.find(x=>x.k===k);const v=s[k];if(f&&f.t==='num')return v==null?-Infinity:v;return String(v==null?'':v).toLowerCase()}
function vLektoren(){
  const q=F.q.toLowerCase();
  let l=SPEAKERS.filter(s=>(F.prio==='alle'||String(s.prio||'').charAt(0)===F.prio)&&(F.list==='alle'||s.list===F.list)&&(F.type==='alle'||s.type===F.type)&&(F.con==='alle'||s.con===F.con)&&(F.area==='alle'||s.main===F.area)&&(F.stat==='alle'||s.v===F.stat)&&SF.map(f=>s[f.k]).join(' ').toLowerCase().includes(q));
  const k=XL.sort.k,d=XL.sort.dir;
  l=[...l].sort((a,b)=>{const x=sortVal(a,k),y=sortVal(b,k);return (x<y?-1:x>y?1:0)*d});
  const hourly=SPEAKERS.filter(s=>s.rate&&(!s.bill||s.bill==='Stundensatz'));const avg=hourly.length?Math.round(hourly.reduce((a,s)=>a+s.rate,0)/hourly.length):0;
  const act=SPEAKERS.filter(s=>s.con==='rahmen').length,free=SPEAKERS.filter(s=>s.v==='Verfügbar').length,self=SPEAKERS.filter(s=>s.origin==='Selbstauskunft').length;
  const sel=(id,cur,all,list)=>'<select id="'+id+'" aria-label="'+all+'"><option value="alle">'+all+'</option>'+list.map(x=>'<option '+(cur===x?'selected':'')+'>'+esc(x)+'</option>').join('')+'</select>';
  const selKv=(id,cur,all,list)=>'<select id="'+id+'" aria-label="'+all+'"><option value="alle">'+all+'</option>'+list.map(([v,t])=>'<option value="'+v+'" '+(cur===v?'selected':'')+'>'+esc(t)+'</option>').join('')+'</select>';

  if(!SPEAKERS.length)return head('Pool','Speaker und Lektoren','',btns)+emptyCard('Noch keine Profile. Importieren Sie die vorbereitete Liste, lassen Sie Lektor:innen das Onboarding-Formular ausfüllen oder legen Sie ein Profil von Hand an.',`<button class="btn" data-seedlek>Lektoren-Liste importieren</button><button class="btn ghost" data-new="speaker">Profil von Hand anlegen</button>`);
  return head('Pool','Speaker und Lektoren',`${SPEAKERS.length} Profile · ${act} mit Rahmenvertrag · ${free} verfügbar${avg?' · Ø Stundensatz '+avg+' €':''}${self?' · '+self+' per Selbstauskunft':''}`,btns)+menu+
  `<div class="tools"><input type="search" id="q" placeholder="Suchen in allen Feldern" value="${esc(F.q)}" aria-label="Suchen">${sel('sel-area',F.area,'Alle Fachgebiete',VOC.hauptfach)}${sel('sel-stat',F.stat,'Jeder Status',VOC.status)}${sel('sel-prio',F.prio,'Jede Prio',['A','B','C'])}${selKv('sel-type',F.type,'Alle Typen',O(['Lektor','Speaker','Beides']))}${selKv('sel-con',F.con,'Jeder Vertrag',CON_OPTS)}${sel('sel-list',F.list,'Alle Listen',lists)}<span class="muted small" style="align-self:center">${l.length} von ${SPEAKERS.length} · Zelle anklicken zum Bearbeiten</span></div>`+

}

/* Bearbeiten direkt in der Zelle */
let XED=null;
function startEdit(td){
  endEdit(true);
  const tr=td.closest('tr');if(!tr||!tr.dataset.id)return;
  const s=SPEAKERS.find(x=>x.id===tr.dataset.id),f=SF.find(x=>x.k===td.dataset.k);if(!s||!f)return;
  const r=td.getBoundingClientRect();
  let el;
  if(f.o){el=document.createElement('select');el.innerHTML='<option value=""></option>'+f.o.map(([v,t])=>`<option value="${esc(v)}">${esc(t)}</option>`).join('');if(s[f.k]&&!f.o.some(x=>x[0]===s[f.k]))el.innerHTML+=`<option value="${esc(s[f.k])}">${esc(s[f.k])}</option>`;el.value=s[f.k]||''}
  else if(f.long){el=document.createElement('textarea');el.value=s[f.k]||''}
  else{el=document.createElement('input');el.type=f.t==='num'?'number':'text';el.value=s[f.k]==null?'':s[f.k]}
  el.className='xe';
  el.style.left=Math.max(4,Math.min(r.left,innerWidth-(f.long?340:r.width)-8))+'px';el.style.top=r.top+'px';
  el.style.width=(f.long?Math.max(r.width,340):r.width)+'px';el.style.height=(f.long?130:r.height)+'px';
  document.body.appendChild(el);el.focus();if(el.select&&!f.o)el.select();
  XED={el,s,f,td,tr};
  el.addEventListener('keydown',ev=>{
    if(ev.key==='Escape'){endEdit(false)}
    else if(ev.key==='Enter'&&!(f.long&&ev.shiftKey)){ev.preventDefault();endEdit(true)}
    else if(ev.key==='Tab'){ev.preventDefault();const cells=[...tr.querySelectorAll('td[data-k]')];const i=cells.indexOf(td);endEdit(true);const nx=cells[i+(ev.shiftKey?-1:1)];if(nx)startEdit(nx)}
  });
  el.addEventListener('blur',()=>setTimeout(()=>{if(XED&&XED.el===el)endEdit(true)},120));
}
async function endEdit(save){
  if(!XED)return;
  const {el,s,f,td}=XED;XED=null;const v=el.value;el.remove();
  if(!save)return;
  const nv=f.t==='num'?(v===''?null:+v):v.trim();
  if(String(s[f.k]==null?'':s[f.k])===String(nv==null?'':nv))return;
  s[f.k]=nv;td.innerHTML=cellHtml(s,f);td.title=nv==null?'':nv;td.classList.add('saved');setTimeout(()=>td.classList.remove('saved'),900);
  await persist('Speaker',dSp(s));
}
