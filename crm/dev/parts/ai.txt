function renderAi(){
  const body=$('#ai-body');if(!body)return;
  if(!CFG.claude){
    body.innerHTML=`<div class="ai-msg a"><p>Damit ich hier mit Ihnen arbeiten kann, brauche ich einen Anthropic-API-Schlüssel. Er wird nur in Ihrem Google-Konto (Skripteigenschaften) gespeichert. Die Nutzung wird nach Verbrauch abgerechnet, ein Auftrag kostet wenige Cent.</p></div><div class="fld"><label for="ai-key">API-Schlüssel (sk-ant-…)</label><input type="password" id="ai-key" autocomplete="off"></div><div class="btns"><button class="btn sm" data-aikey>Schlüssel speichern</button></div>`;
    $('#ai-form').hidden=true;return;
  }
  $('#ai-form').hidden=false;
  const m=AI.msgs;
  body.innerHTML=(m.length?'':`<div class="ai-msg a"><p>Ich bin Claude (Sonnet 5.5) und pflege mit Ihnen das CRM. Schreiben Sie einfach, was sich ändern soll. Ich ordne es der richtigen Person zu und zeige Ihnen, was ich getan habe. Sie können es mit einem Klick rückgängig machen.</p><div class="ai-hints">${AI_HINTS.map((h,i)=>`<button class="chip" data-aihint="${i}">${esc(h)}</button>`).join('')}</div></div>`)+
  m.map((x,i)=>x.role==='user'?`<div class="ai-msg u"><p>${esc(x.text)}</p></div>`:`<div class="ai-msg a"><p>${esc(x.text)}</p>${(x.applied||[]).length?`<ul class="ai-chg">${x.applied.map(a=>`<li><b>${esc(a.label||a.id)}</b> <span class="muted small">${a.op==='create'?'neu angelegt':'aktualisiert'}</span><div class="small">${Object.keys(a.fields).map(k=>`${esc(SERVER_LABEL[k]||k)}: ${esc(String(a.fields[k]).slice(0,60))}`).join(' · ')}</div></li>`).join('')}</ul>${x.undone?'<span class="muted small">rückgängig gemacht</span>':`<button class="btn ghost sm" data-aiundo="${i}">Rückgängig</button>`}`:''}</div>`).join('')+
  (AI.busy?'<div class="ai-msg a"><p class="muted">Claude arbeitet …</p></div>':'');
  body.scrollTop=body.scrollHeight;
}
function toggleAi(force){
  const p=$('#ai');const open=force===undefined?p.hidden:force;p.hidden=!open;document.body.classList.toggle('ai-open',open);
  if(open){renderAi();setTimeout(()=>{const i=$('#ai-in');if(i)i.focus()},60)}
}
async function sendAi(){
  const inp=$('#ai-in');const text=inp.value.trim();if(!text||AI.busy)return;
  inp.value='';AI.msgs.push({role:'user',text});AI.busy=true;renderAi();
  try{
    const history=AI.msgs.slice(0,-1).map(x=>({role:x.role==='user'?'user':'assistant',text:x.text}));
    const r=await srv('api_assistant',text,history);
    AI.msgs.push({role:'assistant',text:r.reply||'Erledigt.',applied:r.applied||[]});
    if((r.applied||[]).length){await reload();rerender()}
  }catch(e){
    const msg=String(e&&e.message||e);
    if(msg.indexOf('NO_KEY')>=0){CFG.claude=false}
    AI.msgs.push({role:'assistant',text:msg.indexOf('NO_KEY')>=0?'Es ist noch kein API-Schlüssel hinterlegt.':'Das hat nicht geklappt: '+msg});
  }
  AI.busy=false;renderAi();
}
