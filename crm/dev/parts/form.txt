let SPDRAFT=null;
const SGROUPS={1:'Stammdaten',2:'Ausbildung und fachliche Spezialisierung',3:'Wünsche zu Honorar und Verfügbarkeit (Angaben der Person)',4:'Vertrag und Status (Ihre Pflege)'};
function sfield(f,s){
  const v=s[f.k]==null?'':s[f.k];
  if(f.o)return fld('sf-'+f.k,f.l,v,{t:'select',opts:[['','—'],...f.o]});
  if(f.long)return fld('sf-'+f.k,f.l,v,{t:'textarea'});
  return fld('sf-'+f.k,f.l,v,{t:f.t==='num'?'number':f.t==='email'?'email':'text'});
}
function openSpeaker(id){
  const s=SPDRAFT||(id?JSON.parse(JSON.stringify(SPEAKERS.find(x=>x.id===id))):{id:'',type:'Lektor',con:'keiner',for:[],sess:[],list:'Eigene Eingabe',origin:'Manuell'});
  SPDRAFT=s;
  let html=`<button class="x" data-close>Schließen</button><div class="eyebrow">${s.id?esc(s.type||'Profil'):'Neu'}</div><h2>${esc(s.n)||'Neues Profil'}</h2>`;
  [1,2,3,4].forEach(g=>{
    const fs=SF.filter(f=>f.g===g);
    html+=`<div class="fld"><span class="l" style="margin-top:14px;color:var(--brand-d)">${SGROUPS[g]}</span></div>`;
    for(let i=0;i<fs.length;){
      if(fs[i].long||fs[i].k==='n'){html+=sfield(fs[i],s);i++;continue}
      const a=fs[i],b=fs[i+1];
      if(b&&!b.long){html+=`<div class="two">${sfield(a,s)}${sfield(b,s)}</div>`;i+=2}else{html+=sfield(a,s);i++}
    }
    if(g===4)html+=`<div class="fld"><span class="l">Einsetzbar für Produkte</span><div class="chips">${PRODUCTS.map(p=>`<button type="button" class="chip" data-sprod="${p.id}" aria-pressed="${s.for.includes(p.id)}">${esc(p.n)}</button>`).join('')}</div></div>${fld('sf-sess','Bisherige Einsätze (ein Eintrag pro Zeile)',(s.sess||[]).join('\n'),{t:'textarea'})}${s.src?`<div class="fld"><span class="l">Quellen</span><div class="small muted" style="overflow-wrap:anywhere">${esc(s.src)}</div></div>`:''}`;
  });
  html+=`<div class="btns"><button class="btn" data-spsave>Speichern</button><button class="btn ghost" data-close>Abbrechen</button>${s.id?delBtn('speaker',s.id):''}</div>`;
  $('#sheet').innerHTML=html;
  $('#scrim').hidden=false;
}
function readSp(){
  const s=SPDRAFT;if(!s||!$('#f-sf-n'))return s;
  SF.forEach(f=>{const e=document.getElementById('f-sf-'+f.k);if(!e)return;const v=e.value.trim();s[f.k]=f.t==='num'?(v===''?null:+v):v});
  const se=document.getElementById('f-sf-sess');if(se)s.sess=se.value.split('\n').map(x=>x.trim()).filter(Boolean);
  return s;
}
async function saveSp(){
  const s=readSp();if(!s.n){toast('Bitte einen Namen eintragen.');return}
  const old=s.id?SPEAKERS.find(x=>x.id===s.id):null;
  if(old)Object.assign(old,s);else{s.id=nid();SPEAKERS.push(s)}
  SPDRAFT=null;closeSheet();rerender();await persist('Speaker',dSp(s));toast('Profil gespeichert');
}
